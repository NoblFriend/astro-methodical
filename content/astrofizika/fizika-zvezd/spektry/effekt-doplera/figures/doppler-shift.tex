% Волновые фронты движущегося источника.
% Компилируется отдельно: xelatex doppler-shift.tex
\documentclass[tikz]{standalone}
\usetikzlibrary{arrows.meta}
\usepackage{fontspec}
\setmainfont{PT Sans}
\begin{document}
\begin{tikzpicture}[scale=1.05]
  % фронты, испущенные в прошлые моменты: центры отстают от источника
  \foreach \i in {1,...,4} {
    \draw[gray!75, thick] ({0.7 - 0.28*\i}, 0) circle ({0.55*\i});
  }
  % источник и его скорость
  \fill (0.7,0) circle (0.07);
  \draw[-{Stealth}, very thick] (0.8,0) -- (1.7,0) node[midway, above=2pt] {$v$};
  \node[below=6pt] at (0.7,-0.1) {источник};
  % наблюдатели
  \node[right, align=left] at (2.6, 0.9) {наблюдатель A\\ (синее смещение)};
  \node[left, align=right] at (-2.9, 0.9) {наблюдатель B\\ (красное смещение)};
\end{tikzpicture}
\end{document}
