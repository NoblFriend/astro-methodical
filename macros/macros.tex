% Единый источник кастомных команд.
% Эти команды работают и в формулах на сайте (KaTeX), и в PDF (XeLaTeX),
% и внутри TikZ-картинок.
%
% Ограничение: чтобы команда подхватилась KaTeX, определяйте её через
% \newcommand{\имя}{...} или \newcommand{\имя}[число аргументов]{...}
% (одна команда — одна строка или один блок с балансом скобок).

% --- Астрономические обозначения ---
\newcommand{\Msun}{M_{\odot}}
\newcommand{\Rsun}{R_{\odot}}
\newcommand{\Lsun}{L_{\odot}}
\newcommand{\Mearth}{M_{\oplus}}
\newcommand{\au}{\text{а.е.}}
\newcommand{\pc}{\text{пк}}

% --- Математика ---
% Пакет physics подключён и на сайте, и в PDF: \dv, \pdv, \abs, \norm,
% \qty, \dd, \vb, \grad и т.д. работают из коробки — здесь их НЕ переопределяем.
\newcommand{\dif}{\mathrm{d}}
\newcommand{\avg}[1]{\left\langle #1 \right\rangle}
\newcommand{\vect}[1]{\boldsymbol{#1}}
